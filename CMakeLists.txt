cmake_minimum_required(VERSION 3.22)
project(GoodLookinUI VERSION 0.2.0 LANGUAGES CXX)
add_library(goodlookinui INTERFACE)
target_compile_features(goodlookinui INTERFACE cxx_std_20)
target_include_directories(goodlookinui INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
# The core can configure without downloading or linking JUCE.
if(TARGET juce::juce_gui_basics)
  add_library(goodlookinui_juce INTERFACE)
  target_link_libraries(goodlookinui_juce INTERFACE goodlookinui juce::juce_gui_basics)
  target_include_directories(goodlookinui_juce INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/adapters/juce)
endif()
option(GOODLOOKINUI_BUILD_TESTS "Build dependency-free core checks" OFF)
if(GOODLOOKINUI_BUILD_TESTS)
  enable_testing()
  find_package(Threads REQUIRED)
  foreach(t core spectrum level_tracker look_table)
    add_executable(goodlookinui_${t} tests/${t}.cpp)
    target_link_libraries(goodlookinui_${t} PRIVATE goodlookinui Threads::Threads)
    target_compile_options(goodlookinui_${t} PRIVATE -O2)
    add_test(NAME goodlookinui_${t} COMMAND goodlookinui_${t})
  endforeach()
endif()
